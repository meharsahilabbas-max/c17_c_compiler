cmake_minimum_required(VERSION 3.20)
project(mycc VERSION 0.1.0 LANGUAGES C)

option(MYCC_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(MYCC_ENABLE_COVERAGE "Enable compiler coverage instrumentation" OFF)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

add_library(mycc-core STATIC src/main.c src/preprocessor.c src/types.c src/symbols.c src/ir.c src/optimizer.c)
target_compile_definitions(mycc-core PRIVATE main=mycc_compile_main)
target_include_directories(mycc-core PRIVATE include)
add_executable(mycc src/driver.c)
target_link_libraries(mycc PRIVATE mycc-core)
if(MSVC)
  target_compile_options(mycc-core PRIVATE /W4)
  target_compile_options(mycc PRIVATE /W4)
else()
  target_compile_options(mycc-core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  target_compile_options(mycc PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(MYCC_ENABLE_SANITIZERS AND NOT MSVC)
  target_compile_options(mycc-core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(mycc PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(mycc-core PRIVATE -fsanitize=address,undefined)
  target_link_options(mycc PRIVATE -fsanitize=address,undefined)
endif()
if(MYCC_ENABLE_COVERAGE AND NOT MSVC)
  target_compile_options(mycc-core PRIVATE --coverage)
  target_compile_options(mycc PRIVATE --coverage)
  target_link_options(mycc-core PRIVATE --coverage)
  target_link_options(mycc PRIVATE --coverage)
endif()

if(WIN32)
  add_executable(mycc-ide WIN32 src/gui.c)
  target_link_libraries(mycc-ide PRIVATE comctl32)
  if(MSVC)
    target_compile_options(mycc-ide PRIVATE /W4)
  else()
    target_compile_options(mycc-ide PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  set_target_properties(mycc-ide PROPERTIES OUTPUT_NAME "mycc-ide")
endif()

include(CTest)
if(BUILD_TESTING)
  add_test(NAME cli_help COMMAND mycc --help)
  add_executable(mycc-ir-test tests/ir_optimizer_test.c)
  target_link_libraries(mycc-ir-test PRIVATE mycc-core)
  target_include_directories(mycc-ir-test PRIVATE include)
  if(MYCC_ENABLE_SANITIZERS AND NOT MSVC)
    target_compile_options(mycc-ir-test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(mycc-ir-test PRIVATE -fsanitize=address,undefined)
  endif()
  if(MYCC_ENABLE_COVERAGE AND NOT MSVC)
    target_compile_options(mycc-ir-test PRIVATE --coverage)
    target_link_options(mycc-ir-test PRIVATE --coverage)
  endif()
  add_test(NAME ir_optimizer COMMAND mycc-ir-test)
  set(MYCC_RUNTIME_SUFFIX "${CMAKE_EXECUTABLE_SUFFIX}")
  add_test(NAME for_loop COMMAND mycc ${CMAKE_CURRENT_SOURCE_DIR}/tests/for_loop.c -o ${CMAKE_CURRENT_BINARY_DIR}/for_loop${MYCC_RUNTIME_SUFFIX})
  add_test(NAME runtime_for_compile COMMAND mycc ${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime_for.c -o ${CMAKE_CURRENT_BINARY_DIR}/runtime_for${MYCC_RUNTIME_SUFFIX})
  add_test(NAME runtime_for_run COMMAND ${CMAKE_CURRENT_BINARY_DIR}/runtime_for${MYCC_RUNTIME_SUFFIX})
  set_tests_properties(runtime_for_run PROPERTIES DEPENDS runtime_for_compile WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
  set_tests_properties(runtime_for_run PROPERTIES DEPENDS runtime_for_compile)
endif()

install(TARGETS mycc RUNTIME DESTINATION bin)
if(WIN32)
  install(TARGETS mycc-ide RUNTIME DESTINATION bin)
endif()
